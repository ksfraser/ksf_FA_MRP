## 3. Test Cases

### TC-001: Single-Level Explosion

| Property | Value |
|----------|-------|
| Test ID | TC-001 |
| Test Name | Single-Level Explosion |
| Priority | HIGH |
| Pre-condition | BOM exists for PRODUCT-A with COMP-B (qty=2) and COMP-C (qty=3) |
| Input | 5 units of PRODUCT-A |
| Expected Output | COMP-B: 10 units, COMP-C: 15 units |

```php
/**
 * @BABOK Related: FR-MRP-001-003, AC-003-01, AC-003-02, AC-003-06
 */
public function testSingleLevelExplosion(): void
{
    $this->bomDAO->method('getBOMItems')
        ->with('PRODUCT-A')
        ->willReturn([
            ['component' => 'COMP-B', 'quantity' => 2, 'is_phantom' => 0],
            ['component' => 'COMP-C', 'quantity' => 3, 'is_phantom' => 0]
        ]);

    $demands = [
        ['item_code' => 'PRODUCT-A', 'quantity' => 5, 'due_date' => '2024-01-15']
    ];

    $result = $this->exploder->explode($demands);

    $this->assertCount(2, $result);

    $components = array_column($result, 'item_code');
    $this->assertContains('COMP-B', $components);
    $this->assertContains('COMP-C', $components);

    $compB = array_filter($result, fn($r) => $r['item_code'] === 'COMP-B');
    $compC = array_filter($result, fn($r) => $r['item_code'] === 'COMP-C');

    $this->assertEquals(10, reset($compB)['gross_qty']);
    $this->assertEquals(15, reset($compC)['gross_qty']);
}
```

### TC-006: Maximum Depth Enforcement

| Property | Value |
|----------|-------|
| Test ID | TC-006 |
| Test Name | Maximum Depth Enforcement |
| Priority | HIGH |
| Pre-condition | BOM chain of 25 items exceeds max depth |
| Input | 1 unit of ITEM-1 |
| Expected Output | Only items up to depth 20 included |

```php
/**
 * @BABOK Related: FR-MRP-001-003, AC-003-15
 */
public function testMaxDepthEnforced(): void
{
    $bomItems = [];
    for ($i = 1; $i <= 25; $i++) {
        $bomItems['ITEM-' . $i] = [
            ['component' => 'ITEM-' . ($i + 1), 'quantity' => 1, 'is_phantom' => 0]
        ];
    }
    $bomItems['ITEM-25'] = [];
    $bomItems['ITEM-26'] = [];

    $this->bomDAO->method('getBOMItems')
        ->willReturnCallback(function ($code) use ($bomItems) {
            return $bomItems[$code] ?? [];
        });

    $demands = [
        ['item_code' => 'ITEM-1', 'quantity' => 1, 'due_date' => '2024-01-15']
    ];

    $result = $this->exploder->explode($demands);

    $this->assertCount(1, $result);
    $this->assertEquals('ITEM-21', $result[0]['item_code']);
}
```

### TC-007: Circular Reference Detection

| Property | Value |
|----------|-------|
| Test ID | TC-007 |
| Test Name | Circular Reference Detection |
| Priority | HIGH |
| Pre-condition | A → B → C → A (circular) |
| Input | 1 unit of ITEM-A |
| Expected Output | No infinite loop, items processed without repetition |

```php
/**
 * @BABOK Related: FR-MRP-001-003, AC-003-16
 */
public function testCircularReferenceDetection(): void
{
    $this->bomDAO->method('getBOMItems')
        ->willReturnCallback(function ($code) {
            if ($code === 'ITEM-A') {
                return [['component' => 'ITEM-B', 'quantity' => 1, 'is_phantom' => 0]];
            }
            if ($code === 'ITEM-B') {
                return [['component' => 'ITEM-C', 'quantity' => 1, 'is_phantom' => 0]];
            }
            if ($code === 'ITEM-C') {
                return [['component' => 'ITEM-A', 'quantity' => 1, 'is_phantom' => 0]];
            }
            return [];
        });

    $this->bomDAO->method('isPhantomItem')
        ->willReturn(false);

    $demands = [
        ['item_code' => 'ITEM-A', 'quantity' => 1, 'due_date' => '2024-01-15']
    ];

    $startTime = microtime(true);
    $result = $this->exploder->explode($demands);
    $endTime = microtime(true);

    $this->assertLessThan(1.0, $endTime - $startTime, 'Explosion took too long - possible infinite loop');
}
```
